# Native app shell — WKWebView on macOS, no browser, no HTTP server.
# The UI loads embedded HTML directly via pywebview's html= parameter and

pywebview>=4.4

# Text-to-speech (cross-platform). On Linux you can instead install the
# system package espeak-ng and skip pyttsx3.

# macOS only — used to enumerate cameras via AVFoundation (the same API
# OpenCV uses on macOS 14+), so we can reliably skip Continuity Camera
# (iPhone-as-webcam). Without it we fall back to ffmpeg/system_profiler,
# which lie when Continuity is active. PyObjC's Cocoa + WebKit frameworks
# are pulled in transitively by pywebview on macOS.
pyobjc-framework-AVFoundation>=10.0; sys_platform == "darwin"
